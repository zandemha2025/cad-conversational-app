## EXAMPLE — Complete Drill Jig:

```python
import cadquery as cq

# Parameters
plate_length = 200  # mm
plate_width = 150
plate_thick = 15
hole_dia = 8.0  # drill bushing bore
cbore_dia = hole_dia * 1.25  # counterbore for bushing seat
cbore_depth = 8.0
grid_cols = 2
grid_rows = 2
grid_spacing_x = 50.0
grid_spacing_y = 50.0

# Bushing hole positions (centered on plate)
bushing_pts = []
for col in range(grid_cols):
    for row in range(grid_rows):
        x = (col - (grid_cols - 1) / 2) * grid_spacing_x
        y = (row - (grid_rows - 1) / 2) * grid_spacing_y
        bushing_pts.append((x, y))

# Base plate
result = cq.Workplane('XY').box(plate_length, plate_width, plate_thick)

# Drill bushing counterbore holes
result = (result.faces('>Z').workplane()
    .pushPoints(bushing_pts)
    .cboreHole(hole_dia, cbore_dia, cbore_depth))

# Corner mounting holes (28mm in from edges)
mount_sx = plate_length - 56  # 28mm margin each side
mount_sy = plate_width - 56
result = (result.faces('>Z').workplane()
    .rect(mount_sx, mount_sy, forConstruction=True)
    .vertices().hole(8))

# Datum A: 3 support pads (12mm dia, 3mm tall)
pad_pts = [
    (-plate_length/2 + 30, -plate_width/2 + 30),
    (plate_length/2 - 30, -plate_width/2 + 30),
    (0, plate_width/2 - 30),
]
result = (result.faces('>Z').workplane()
    .pushPoints(pad_pts)
    .circle(6).extrude(3))

# Datum B: 2 locating pins along long edge (6mm dia, 12mm tall)
pin_b_pts = [(-plate_length/4, -plate_width/2 + 20), (plate_length/4, -plate_width/2 + 20)]
result = (result.faces('>Z').workplane(offset=3)
    .pushPoints(pin_b_pts)
    .circle(3).extrude(12))

# Datum C: 1 stop pin on short edge
result = (result.faces('>Z').workplane(offset=3)
    .center(-plate_length/2 + 20, 0)
    .circle(3).extrude(12))

# Clamp slots (2 slots on opposite long edges)
result = (result.faces('>Z').workplane()
    .center(0, plate_width/2 - 15)
    .slot2D(60, 12).cutThruAll())
result = (result.faces('>Z').workplane()
    .center(0, -plate_width/2 + 15)
    .slot2D(60, 12).cutThruAll())

# Edge chamfers
result = result.edges('|Z').chamfer(2)

cq.exporters.export(result, '/tmp/cq_output.step')
```
